class Solution {
    public int[] twoSum(int[] arr, int target) {
        HashMap<Integer,Integer> hm=new HashMap<>();

        int ans[]=new int[2];
        ans[0]=ans[1]=-1;

        for(int i=0;i<arr.length;i++){
            int need=target-arr[i];
            if(hm.containsKey(need)){
                ans[0]=hm.get(need);
                ans[1]=i;
                return ans;
            }else{
                hm.put(arr[i],i);
            }
        }
        return ans;

    }
}
